Get-ChildItem .\Output\Live_Exports -File | Where-Object { $_.Name -match '^(README|BUDGET|NOTES)' } | Get-FileHash -Algorithm SHA256 | Select-Object Hash,@{n='File';e={Split-Path $_.Path -Leaf}}

Get-ChildItem .\Output\Deleted_Export -File | Where-Object { $_.Name -match 'RANSFER' } | Get-FileHash -Algorithm SHA256 | Select-Object Hash,@{n='File';e={Split-Path $_.Path -Leaf}}

STEP 9  Records and hash manifest

Copy-Item '.\Output\Raw\USB001.001.txt' .\Output\Records\

$hashes | Export-Csv .\Output\Records\Output_Hashes.csv -NoTypeInformation
